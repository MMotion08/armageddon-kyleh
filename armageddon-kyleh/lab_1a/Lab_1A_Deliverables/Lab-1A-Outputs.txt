Verify IAM Role Attached to an EC2 Instance
  Step 1: Identify the EC2 instance

aws ec2 describe-instances \
      --filters Name=tag:Name,Values=MyInstance \
      --region us-east-2 \
      --query "Reservations[].Instances[].InstanceId" \
      --output text


Step 2: Check the IAM role attached to the instance

aws ec2 describe-instances \
      --instance-ids i-02e2b700b5e1a6b5d \
      --region us-east-2 \
      --query "Reservations[].Instances[].IamInstanceProfile.Arn" \
      --output text
arn:aws:iam::912083617270:instance-profile/lab1a-ec2-secrets-role

Step 3: Resolve instance profile → role name

aws iam get-instance-profile \
      --instance-profile-name lab1a-ec2-secrets-role \
      --query "InstanceProfile.Roles[].RoleName" \
      --output text
lab1a-ec2-secrets-role
